import express from 'express';
import db from './repository/index.js';
 
const app = express() -- Sempre no inicio do programa
app.use(express.json()) -- Sempre no inicio do programa
 
//C
app.post("/api/v1/pessoa", (req, res) => {
    //id, cpf, email
    const { id, cpf, email } = req.body
 
    if (!id || !cpf || !email) {
        //avisar q é para enviar tudo
        res.send({ message: "Falta de informação" })
        return
    }
    //cadastrar
    db.push({ id, cpf, email })
    res.send({ message: "Pessoa criada com sucesso!" })
})

app.get("/api/v1/pessoas", (req, res) => {
    res.send({message: "Pessoa encontrada com sucesso", db})
 })

//R
app.get("/api/v1/pessoa/:id", (req, res) => {
    const id = Number(req.params.id)

    const pessoa = db.find(p => p.id === id)

    if(!pessoa) {
        res.status(404).send({message: "Pessoa não encontrada!"})
        return
    }
    res.send({message: "Pessoa encontrada com sucesso", pessoa})
 })
//U
app.put("/api/v1/pessoa/:id", (req, res) => { 
    const id = Number(req.params.id)
    const {email} = req.body
    const pessoa = db.find(it => it.id === id)

    if(!pessoa) {
        res.status(404).send({message: "Pessoa não encontrada!"})
        return
    }
    
    pessoa.email = email
    res.send({message: "Pessoa alterada com sucesso"})
})
//D
app.delete("/api/v1/pessoa/:id", (req, res) => {
    const id = Number(req.params.id)

    const index = db.findIndex(it => it.id === id)

    if(index === -1) {
        res.status(404).send({message: "Pessoa não encontrada"})
        return
    }

    db.splice(index, 1)

    res.send({message: "Pessoa deletada com sucesso!"})

 })
 
app.listen(3000, () => {
    console.log(`Servidor rodando em http://localhost:${3000}`);
});


// res.send() → envia resposta com status padrão 200,
// res.status(404).send() → envia resposta com status 404
// p.id e it.id → não têm diferença; são apenas nomes diferentes para o mesmo elemento percorrido pelo find().

Postman 


Get - pega as informacões
Post - Enviar as informacões (Cria)
Put - Atualiza as informacões
Delete - Deleta as informacões


status 200 indica que a requisição feita pelo cliente ao servidor foi bem-sucedida

status 404 indica que o navegador conseguiu se comunicar com o servidor, mas o servidor não conseguiu encontrar o recurso solicitado -> geralmente é porque não tem o dado em especifico ou porque não encotrou a rota que você está tentando acessar

status 500 indica que o servidor encontrou uma condição inesperada e não conseguiu concluir a solicitação do usuário


Findall = Pegar todos 

Metodo GET 
Utiliza apenas a rota pegar todos não tendo necessidade do indentificador 

exemplo:
GET -> localhost:3000/api/v1/pessoas



FindOne = Pegar um 

Metodo GET 
Utiliza a rota pegar e tem necessidade do indentificador 

exemplo:
GET -> localhost:3000/api/v1/pessoa/1 
o identificado vai variar pela posição que está armazenado no banco de dados "vetor"


Create = criar

Metodo POST
Utiliza a rota e dentro do corpo da requisição tem o objeto que vai as informaçoes que são solictadas 

exemplo:
POST -> localhost:3000/api/v1/pessoa

body 
raw -- json

{
    id: 1,
    cpf: "12345678901",
    email: "caio@email.com"
}



Update = Atualizar

Metodo PUT
Utiliza a rota e o ID e dentro do corpo da requisição tem o objeto que vai as informaçoes que são solictadas 

exemplo:
PUT -> localhost:3000/api/v1/pessoa/1

body 
raw -- json

{
    cpf: "98765432100",
    email: "andreia@email.com"
}

Antes o GET puxava assim 
id: 1,
cpf: "12345678901",
email: "caio@email.com"

Após a atualização o metodo GET vai puxar assim
id 1,
cpf: "98765432100",
email: "andreia@email.com"



Delete = Deletar

Metodo DELETE
Utiliza a rota pegar e tem necessidade do indentificador 

exemplo:
DELETE -> localhost:3000/api/v1/pessoa/1 
o identificado vai variar pela posição que está armazenado no banco de dados "vetor"
